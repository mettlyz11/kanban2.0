```sql
    date(timestamp) as date,

    SUM(cost_usd) as total_cost_usd,
```

### 4.4 向量化Pipeline

```python
def vectorize_memory_files(memory_dir: str):
    """批量向量化记忆文件"""
    import sqlite_vec
    import sqlite3
    from sentence_transformers import SentenceTransformer
    
    model = SentenceTransformer('all-MiniLM-L6-v2')
    conn = sqlite3.connect('~/.openclaw/observability/memory_vectors.db')
    conn.enable_load_extension(True)
    sqlite_vec.load(conn)
    
    # 创建向量表
    conn.execute("""
        CREATE VIRTUAL TABLE memory_chunks USING vec0(
            embedding float[384],
            file_path text,
            chunk_id integer,
            content text
        )
    """)
    
    # 处理所有memory/*.md文件
    for md_file in Path(memory_dir).glob("*.md"):
        chunks = chunk_markdown(md_file, chunk_size=500)
        for i, chunk in enumerate(chunks):
            embedding = model.encode(chunk).tolist()
            conn.execute("""
                INSERT INTO memory_chunks (embedding, file_path, chunk_id, content)
                VALUES (?, ?, ?, ?)
            """, (embedding, md_file.name, i, chunk))
    
    conn.commit()
```

### 5.2 基准测试脚本

```python
#!/usr/bin/env python3
"""
MCP + Observability 性能基准测试
输出: benchmark-results-{date}.json
"""

import time
import json
import subprocess
from datetime import datetime

class BenchmarkSuite:
    def __init__(self):
        self.results = []
    
    def test_mcp_tool_latency(self, tool_name: str, iterations: int = 10):
        """测试MCP工具调用延迟"""
        latencies = []
        for _ in range(iterations):
            start = time.time()
            # 调用MCP工具
            subprocess.run(["npx", "-y", f"@modelcontextprotocol/server-{tool_name}"])
            latencies.append((time.time() - start) * 1000)
        
        return {
            "tool": tool_name,
            "p50": sorted(latencies)[len(latencies)//2],
            "p95": sorted(latencies)[int(len(latencies)*0.95)],
            "p99": sorted(latencies)[int(len(latencies)*0.99)],
            "avg": sum(latencies) / len(latencies),
            "min": min(latencies),
            "max": max(latencies)
        }
    
    def test_memory_retrieval_accuracy(self, queries: list):
        """测试记忆检索准确率"""
        results = []
        for q in queries:
            start = time.time()
            # 调用memory_search
            result = subprocess.run(
                ["python3", "-c", f"import memory_search; print(memory_search('{q}'))"],
                capture_output=True, text=True
            )
            latency = (time.time() - start) * 1000
            results.append({
                "query": q,
                "latency_ms": latency,
                "found": len(result.stdout.strip()) > 0
            })
        return results
    
    def run_all(self):
        """运行全部基准测试"""
        report = {
            "timestamp": datetime.now().isoformat(),
            "mcp_latency": {},
            "memory_accuracy": [],
            "subagent_startup": [],
            "summary": {}
        }
        
        # MCP工具延迟
        for tool in ["github", "filesystem", "tavily"]:
            report["mcp_latency"][tool] = self.test_mcp_tool_latency(tool)
        
        # 记忆检索
        report["memory_accuracy"] = self.test_memory_retrieval_accuracy([
            "T109项目进展",
            "Server 1配置",
            "北航联合实验室"
        ])
        
        # 子代理启动
        for i in range(5):
            start = time.time()
            subprocess.run(["openclaw", "session", "status"])
            report["subagent_startup"].append((time.time() - start) * 1000)
        
        # 总结
        report["summary"] = {
            "mcp_tools_tested": len(report["mcp_latency"]),
            "memory_queries_tested": len(report["memory_accuracy"]),
            "subagent_tests": len(report["subagent_startup"]),
            "overall_status": "PASS" if all(
                r.get("p95", 0) < 5000 
                for r in report["mcp_latency"].values()
            ) else "NEEDS_OPTIMIZATION"
        }
        
        # 保存报告
        filename = f"benchmark-results-{datetime.now().strftime('%Y%m%d-%H%M')}.json"
        with open(filename, "w") as f:
            json.dump(report, f, indent=2, ensure_ascii=False)
        
        return report
```

## 八、关键决策点

### 决策1: MCP Server部署方式
- **方案A**: Stdio协议（本地进程，低延迟，仅本地）
- **方案B**: SSE协议（远程可访问，高延迟）
- **推荐**: 混合（本地Stdio + 远程SSE）

### 决策2: 向量嵌入模型
- **方案A**: all-MiniLM-L6-v2（384维，轻量，离线）
- **方案B**: text-embedding-3-small（1536维，OpenAI API）
- **推荐**: 方案A（离线可用，384维足够记忆检索）

### 决策3: 图存储方案
- **方案A**: SQLite + JSONL（轻量，已有基础）
- **方案B**: Neo4j（专业图数据库，重量级）
- **推荐**: 方案A（当前规模无需Neo4j）

### 决策4: 成本核算精度
- **方案A**: 估算（基于token计数 × 公开价格）
- **方案B**: 精确（对接账单API）
- **推荐**: 方案A（足够日常监控，方案B作为Phase 3补充）
