## 1. 会议自动化工作流配置

### 1.1 录音转文字脚本 (whisper_transcribe.py)

```python
#!/usr/bin/env python3
"""
会议录音自动转写脚本
依赖: pip install openai python-dotenv
"""
import os
import sys
from pathlib import Path
from openai import OpenAI
from dotenv import load_dotenv

load_dotenv()

client = OpenAI(api_key=os.getenv('OPENAI_API_KEY'))

def transcribe_audio(audio_path):
    """转写音频文件"""
    print(f"正在转写: {audio_path}")
    
    with open(audio_path, "rb") as audio_file:
        transcript = client.audio.transcriptions.create(
            model="whisper-1",
            file=audio_file,
            response_format="verbose_json",
            timestamp_granularities=["word", "segment"]
        )
    
    # 保存转写结果
    output_path = Path(audio_path).with_suffix('.md')
    with open(output_path, 'w', encoding='utf-8') as f:
        f.write(f"# 会议录音转写\n\n")
        f.write(f"**文件**: {Path(audio_path).name}\n\n")
        f.write(f"## 转写内容\n\n")
        f.write(transcript.text)
    
    print(f"转写完成: {output_path}")
    return output_path

if __name__ == "__main__":
    if len(sys.argv) > 1:
        transcribe_audio(sys.argv[1])
```

```python
#!/usr/bin/env python3
"""
从转写文本中提取会议结构和行动项
"""
import re
from pathlib import Path
import anthropic

def extract_meeting_notes(transcript_path):
    """提取会议纪要"""
    
    with open(transcript_path, 'r', encoding='utf-8') as f:
        content = f.read()
    
    client = anthropic.Anthropic(api_key=os.getenv('ANTHROPIC_API_KEY'))
    
    prompt = f"""
    请将以下会议转写整理成结构化的会议纪要。
    
    要求输出格式：
    # 会议纪要
    
    ## 基本信息
    - 日期：
    - 参会人：
    
    ## 议题
    （列出讨论的主要议题）
    
    ## 讨论摘要
    （每个议题的讨论要点）
    
    ## 决议事项
    （达成的明确决议）
    
    ## 行动项
    | 任务内容 | 责任人 | 截止日期 | 优先级 |
    |---------|--------|---------|--------|
    | 具体任务 | 姓名 | YYYY-MM-DD | 高/中/低 |
    
    会议转写内容：
    {content}
    """
    
    message = client.messages.create(
        model="claude-3-7-sonnet-20250219",
        max_tokens=4000,
        messages=[{"role": "user", "content": prompt}]
    )
    
    # 保存纪要
    output_path = str(transcript_path).replace('.md', '_纪要.md')
    with open(output_path, 'w', encoding='utf-8') as f:
        f.write(message.content[0].text)
    
    print(f"纪要生成: {output_path}")
    return output_path
```

```python
#!/usr/bin/env python3
"""
邮件智能分类与自动回复生成
"""
import imaplib
import email
from email.header import decode_header
import os
from dotenv import load_dotenv

load_dotenv()

def connect_email():
    """连接邮箱服务器"""
    imap = imaplib.IMAP4_SSL("imap.gmail.com")
    imap.login(os.getenv('EMAIL_USER'), os.getenv('EMAIL_PASSWORD'))
    return imap

def get_unread_emails(limit=10):
    """获取未读邮件"""
    imap = connect_email()
    imap.select("INBOX")
    
    _, messages = imap.search(None, 'UNSEEN')
    email_ids = messages[0].split()[-limit:]
    
    emails = []
    for eid in email_ids:
        _, msg_data = imap.fetch(eid, "(RFC822)")
        msg = email.message_from_bytes(msg_data[0][1])
        
        subject = decode_header(msg["Subject"])[0][0]
        if isinstance(subject, bytes):
            subject = subject.decode()
        
        sender = msg.get("From")
        
        # 获取正文
        body = ""
        if msg.is_multipart():
            for part in msg.walk():
                if part.get_content_type() == "text/plain":
                    body = part.get_payload(decode=True).decode()
                    break
        else:
            body = msg.get_payload(decode=True).decode()
        
        emails.append({
            "id": eid.decode(),
            "subject": subject,
            "sender": sender,
            "body": body[:500]
        })
    
    imap.close()
    imap.logout()
    return emails
```

### 4.2 daily_morning_summary.py 脚本框架

```python
#!/usr/bin/env python3
"""
每日晨间摘要脚本
集成：日历 + Things + 天气 + 邮件
"""
import datetime
import subprocess
import requests

def get_calendar_events():
    """获取今日日历事件（Apple Calendar）"""
    # 使用icalBuddy工具
    result = subprocess.run(
        ['icalBuddy', '-npn', '-b', '• ', 'eventsToday'],
        capture_output=True, text=True
    )
    return result.stdout

def get_things_tasks():
    """获取Things今日待办"""
    # 使用things-cli
    result = subprocess.run(
        ['things-cli', 'today'],
        capture_output=True, text=True
    )
    return result.stdout

def get_weather():
    """获取天气信息"""
    api_key = os.getenv('WEATHER_API_KEY')
    url = f"http://api.weatherapi.com/v1/forecast.json?key={api_key}&q=Beijing&days=1"
    response = requests.get(url)
    data = response.json()
    return f"{data['current']['temp_c']}°C, {data['current']['condition']['text']}"

def send_imessage(message):
    """通过iMessage发送"""
    script = f'''
    tell application "Messages"
        send "{message}" to buddy "+8613800000000" of service "E:your@email.com"
    end tell
    '''
    subprocess.run(['osascript', '-e', script])

if __name__ == "__main__":
    today = datetime.date.today().strftime("%Y年%m月%d日")
    
    summary = f"""
📅 {today} 晨间摘要

📆 今日日程：
{get_calendar_events()}

✅ 今日待办：
{get_things_tasks()}

🌤️ 天气：{get_weather()}

💡 建议：今天重点完成优先级高的3件事！
    """
    
    send_imessage(summary)
    print("晨间摘要已发送！")
```

## 5. 安全配置与授权范围

### 5.1 敏感数据处理原则

#### ✅ 已授权可自动化处理：
- 公开会议的录音和纪要（项目组内）
- 常规事务性邮件（供应商、行政通知）
- 公开文献和网络数据
- 个人笔记和知识管理

#### ⚠️ 需人工审核：
- 客户沟通邮件（生成草稿，人工审核发送）
- 内部管理会议纪要（AI整理后人工确认）
- 财务相关邮件（仅分类提醒，不自动回复）

#### ❌ 禁止自动化处理：
- 法务合同和律师沟通邮件
- 人事任免和敏感HR信息
- 涉及核心商业机密的内容
- 银行账户和支付相关操作

### 5.2 数据安全措施

1. **API密钥管理**: 所有密钥存储在 `~/.openclaw/.env`，永不提交Git
2. **日志脱敏**: 所有自动化日志不包含敏感内容
3. **本地优先**: 尽可能使用本地模型处理敏感数据
4. **权限最小化**: 每个脚本只授予必要的权限

### 7.2 每周效率报告生成脚本

```python
#!/usr/bin/env python3
"""
每周效率提升报告
"""
from lib.db_connector import get_db_connection
import datetime

def generate_weekly_report():
    conn = get_db_connection()
    c = conn.cursor()
    
    # 本周数据
    week_start = datetime.date.today() - datetime.timedelta(days=7)
    
    c.execute("""
        SELECT task_type, 
               AVG(CASE WHEN ai_assisted THEN duration_minutes END) as ai_duration,
               AVG(CASE WHEN NOT ai_assisted THEN duration_minutes END) as manual_duration,
               COUNT(*) as count
        FROM efficiency_metrics
        WHERE created_at >= %s
        GROUP BY task_type
    """, (week_start,))
    
    results = c.fetchall()
    
    report = "# 本周AI效率报告\n\n"
    total_saved = 0
    
    for row in results:
        if row[1] and row[2]:  # 有对比数据
            saved = (row[2] - row[1]) * row[3]
            total_saved += saved
            improvement = (row[2] - row[1]) / row[2] * 100
            report += f"- {row[0]}: 平均节省 {row[2]-row[1]:.1f}分钟/任务，提升 {improvement:.0f}%\n"
    
    report += f"\n**本周累计节省: {total_saved:.0f}分钟 ≈ {total_saved/60:.1f}小时**"
    
    conn.close()
    return report
```
