```python
import whisper

def transcribe_audio(audio_path):
    """使用本地Whisper转写录音"""
    model = whisper.load_model("large-v3")
    result = model.transcribe(
        audio_path,
        language="zh",
        word_timestamps=True
    )
    return result["text"], result["segments"]
```

#### Step 4: 任务自动分发
```python
def create_tasks(action_items, meeting_id):
    """自动创建看板任务"""
    from lib.db_connector import get_db_connection

    conn = get_db_connection()
    c = conn.cursor()

    for item in action_items:
        priority_map = {"high": "P0", "medium": "P1", "low": "P2"}
        c.execute('''
            INSERT INTO tasks
            (title, description, status, priority, assignee, due_date,
             category, created_at, updated_at)
            VALUES (%s, %s, %s, %s, %s, %s, %s, NOW(), NOW())
        ''', (
            item["task"][:100],
            f"来源: 会议 #{meeting_id}\n任务详情: {item['task']}",
            "pending",
            priority_map.get(item["priority"], "P1"),
            item["owner"],
            item["deadline"],
            "会议行动项"
        ))

    conn.commit()
    conn.close()
```

```python
def sync_to_zotero(paper, summary):
    """同步到Zotero文献管理工具"""
    from pyzotero import zotero

    zot = zotero.Zotero(ZOTERO_LIBRARY_ID, "user", ZOTERO_API_KEY)

    # 创建条目
    item = zot.item_template("journalArticle")
    item["title"] = paper["title"]
    item["creators"] = [{"creatorType": "author", "name": author}
                       for author in paper["authors"]]
    item["abstractNote"] = summary
    item["tags"] = [paper["topic"], "AI-generated"]

    # 上传PDF附件
    zot.create_items([item])
    zot.attachment_simple([paper["pdf_path"]], item["key"])
```

### 3.2 配置步骤

#### Step 1: 邮件拉取配置
```python
def fetch_unread_emails():
    """拉取未读邮件"""
    import imaplib
    import email

    mail = imaplib.IMAP4_SSL("imap.gmail.com")
    mail.login(EMAIL_USER, EMAIL_PASSWORD)
    mail.select("inbox")

    _, search_data = mail.search(None, "UNSEEN")
    emails = []

    for num in search_data[0].split():
        _, data = mail.fetch(num, "(RFC822)")
        msg = email.message_from_bytes(data[0][1])

        emails.append({
            "id": num.decode(),
            "subject": msg["subject"],
            "from": msg["from"],
            "body": get_email_body(msg),
            "date": msg["date"]
        })

    return emails
```

#### Step 2: 智能分类与优先级判定

```python
def classify_email(email_data):
```

```python
def process_email(email_data):
    """处理单封邮件"""
    classification = classify_email(email_data)
    todos = extract_todos(email_data)

    # 高优先级邮件立即通知
    if classification["priority"] == "high":
        send_notification(
            f"⚠️ 高优先级邮件: {classification['summary']}",
            channel="urgent"
        )

    # 创建待办任务
    for todo in todos:
        create_task_from_todo(todo, source="email")

    return classification, todos
```
