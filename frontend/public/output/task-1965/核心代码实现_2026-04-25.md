```python
#!/usr/bin/env python3
"""
enhanced_task_fsm.py - OpenClaw 增强型任务状态机
实现 P5: State Machine 模式
"""

from enum import Enum
from typing import Optional, Callable, Dict
from datetime import datetime
import json

class TaskState(Enum):
    PENDING = "pending"
    PLANNING = "planning"           # NEW: 正在规划
    EXECUTING = "executing"         # NEW: 正在执行
    VALIDATING = "validating"       # NEW: 正在验证
    RETRY_WAITING = "retry_waiting" # NEW: 等待重试
    RETRYING = "retrying"           # NEW: 正在重试
    VALID_PASSED = "valid_passed"   # NEW: 验证通过
    EXCEEDED_MAX = "exceeded_max"   # NEW: 超过最大重试
    COMPLETED = "completed"
    FAILED = "failed"

class Transition:
    """定义状态转换规则"""
    def __init__(self, from_state: TaskState, to_state: TaskState,
                 condition: Optional[Callable] = None):
        self.from_state = from_state
        self.to_state = to_state
        self.condition = condition
    
    def is_valid(self, context: dict) -> bool:
        if self.condition is None:
            return True
        return self.condition(context)

class TaskStateMachine:
    """任务状态机"""
    
    def __init__(self):
        self.transitions: Dict[tuple, Transition] = {}
        self._define_transitions()
    
    def _define_transitions(self):
        """定义所有合法状态转换"""
        rules = [
            (TaskState.PENDING, TaskState.PLANNING, None),
            (TaskState.PLANNING, TaskState.EXECUTING, 
             lambda ctx: ctx.get('plan') is not None),
            (TaskState.PLANNING, TaskState.FAILED,
             lambda ctx: ctx.get('plan_error') is not None),
            (TaskState.EXECUTING, TaskState.VALIDATING, None),
            (TaskState.EXECUTING, TaskState.RETRY_WAITING,
             lambda ctx: ctx.get('error_type') in RETRYABLE_ERRORS),
            (TaskState.EXECUTING, TaskState.FAILED,
             lambda ctx: ctx.get('error_type') not in RETRYABLE_ERRORS),
            (TaskState.RETRY_WAITING, TaskState.RETRYING, None),
            (TaskState.RETRYING, TaskState.EXECUTING,
             lambda ctx: ctx.get('retry_count', 0) < ctx.get('max_retries', 3)),
            (TaskState.RETRYING, TaskState.EXCEEDED_MAX,
             lambda ctx: ctx.get('retry_count', 0) >= ctx.get('max_retries', 3)),
            (TaskState.VALIDATING, TaskState.VALID_PASSED,
             lambda ctx: ctx.get('validation_passed', False)),
            (TaskState.VALIDATING, TaskState.RETRY_WAITING,
             lambda ctx: not ctx.get('validation_passed', False)),
            (TaskState.VALID_PASSED, TaskState.COMPLETED, None),
            (TaskState.EXCEEDED_MAX, TaskState.FAILED, None),
        ]
        
        for from_s, to_s, condition in rules:
            key = (from_s, to_s)
            self.transitions[key] = Transition(from_s, to_s, condition)
    
    def can_transition(self, current: TaskState, target: TaskState, 
                       context: dict) -> tuple[bool, str]:
        """检查状态转换是否合法"""
        key = (current, target)
        if key not in self.transitions:
            return False, f"非法转换: {current.value} → {target.value}"
        
        transition = self.transitions[key]
        if not transition.is_valid(context):
            return False, f"转换条件不满足: {current.value} → {target.value}"
        
        return True, "OK"
    
    def get_valid_next_states(self, current: TaskState) -> list[TaskState]:
        """获取当前状态的所有合法下一状态"""
        return [t.to_state for (f, t), _ in self.transitions.items() 
                if f == current]


# ==================== 数据库集成 ====================

class TaskStateManager:
    """与数据库集成的状态管理器"""
    
    def __init__(self, db_connection):
        self.fsm = TaskStateMachine()
        self.db = db_connection
    
    def advance_state(self, task_id: int, target_state: TaskState, 
                      context: dict = None) -> dict:
        """推进任务状态"""
        context = context or {}
        
        # 获取当前状态
        task = self._get_task(task_id)
        current = TaskState(task['status'])
        
        # 检查转换合法性
        valid, message = self.fsm.can_transition(current, target_state, context)
        if not valid:
            return {'success': False, 'error': message}
        
        # 更新数据库
        self._update_task_state(task_id, target_state, context)
        
        return {'success': True, 'from': current.value, 'to': target_state.value}
    
    def _get_task(self, task_id: int) -> dict:
        c = self.db.cursor(dictionary=True)
        c.execute('SELECT * FROM tasks WHERE id = %s', (task_id,))
        task = c.fetchone()
        c.close()
        return task
    
    def _update_task_state(self, task_id: int, state: TaskState, context: dict):
        c = self.db.cursor()
        
        updates = ['status = %s', 'updated_at = NOW()']
        params = [state.value]
        
        if state == TaskState.PLANNING:
            updates.append('planning_started_at = NOW()')
        elif state == TaskState.EXECUTING:
            updates.append('started_at = NOW()')
        elif state == TaskState.VALIDATING:
            updates.append('validation_started_at = NOW()')
        elif state == TaskState.COMPLETED:
            updates.append('completed_at = NOW()')
        
        if context.get('plan'):
            updates.append('plan = %s')
            params.append(json.dumps(context['plan']))
        
        if context.get('error'):
            updates.append('last_error = %s')
            params.append(context['error'])
        
        if 'retry_count' in context:
            updates.append('retry_count = %s')
            params.append(context['retry_count'])
        
        if context.get('validation_result'):
            updates.append('validation_result = %s')
            params.append(json.dumps(context['validation_result']))
        
        params.append(task_id)
        sql = f"UPDATE tasks SET {', '.join(updates)} WHERE id = %s"
        c.execute(sql, params)
        self.db.commit()
        c.close()
```

## 模块二: 回滚重试机制 (P8 Rollback & Retry)

```python
#!/usr/bin/env python3
"""
retry_manager.py - OpenClaw 回滚重试管理器
实现 P8: Rollback & Retry 模式
"""

import time
import random
from datetime import datetime, timedelta
from typing import Callable, Optional
from enum import Enum

class ErrorType(Enum):
    NETWORK_TIMEOUT = "network_timeout"
    API_RATE_LIMIT = "api_rate_limit"
    TEMPORARY_CONNECTION_LOSS = "temporary_connection_loss"
    MODEL_PROVIDER_ERROR = "model_provider_error"
    INVALID_TASK_DEFINITION = "invalid_task_definition"
    AUTHENTICATION_FAILURE = "authentication_failure"
    PERMISSION_DENIED = "permission_denied"
    VALIDATION_FAILED = "validation_failed"
    UNKNOWN_ERROR = "unknown_error"

RETRYABLE_ERRORS = {
    ErrorType.NETWORK_TIMEOUT,
    ErrorType.API_RATE_LIMIT,
    ErrorType.TEMPORARY_CONNECTION_LOSS,
    ErrorType.MODEL_PROVIDER_ERROR,
}

NON_RETRYABLE_ERRORS = {
    ErrorType.INVALID_TASK_DEFINITION,
    ErrorType.AUTHENTICATION_FAILURE,
    ErrorType.PERMISSION_DENIED,
}

class RetryPolicy:
    """重试策略配置"""
    
    def __init__(self, max_retries: int = 3,
                 initial_delay: float = 30.0,
                 max_delay: float = 300.0,
                 backoff_multiplier: float = 2.0,
                 jitter: bool = True):
        self.max_retries = max_retries
        self.initial_delay = initial_delay
        self.max_delay = max_delay
        self.backoff_multiplier = backoff_multiplier
        self.jitter = jitter
    
    def calculate_delay(self, attempt: int) -> float:
        """计算下一次重试的延迟时间（指数退避 + 随机抖动）"""
        delay = self.initial_delay * (self.backoff_multiplier ** attempt)
        delay = min(delay, self.max_delay)
        
        if self.jitter:
            # 添加 ±25% 随机抖动，避免惊群效应
            jitter_range = delay * 0.25
            delay += random.uniform(-jitter_range, jitter_range)
        
        return max(delay, 1.0)  # 至少延迟1秒
    
    def should_retry(self, error_type: ErrorType, retry_count: int) -> bool:
        """判断是否应该重试"""
        if retry_count >= self.max_retries:
            return False
        return error_type in RETRYABLE_ERRORS


class RetryManager:
    """重试管理器"""
    
    def __init__(self, policy: Optional[RetryPolicy] = None):
        self.policy = policy or RetryPolicy()
        self.retry_log = []
    
    def classify_error(self, error: Exception) -> ErrorType:
        """将异常分类为 ErrorType"""
        error_str = str(error).lower()
        
        if 'timeout' in error_str or 'timed out' in error_str:
            return ErrorType.NETWORK_TIMEOUT
        if 'rate limit' in error_str or '429' in error_str:
            return ErrorType.API_RATE_LIMIT
        if 'connection' in error_str and 'refused' in error_str:
            return ErrorType.TEMPORARY_CONNECTION_LOSS
        if any(kw in error_str for kw in ['model', 'provider', 'service unavailable']):
            return ErrorType.MODEL_PROVIDER_ERROR
        if any(kw in error_str for kw in ['invalid', 'malformed', 'missing field']):
            return ErrorType.INVALID_TASK_DEFINITION
        if 'auth' in error_str or '401' in error_str or '403' in error_str:
            return ErrorType.AUTHENTICATION_FAILURE
        
        return ErrorType.UNKNOWN_ERROR
    
    def execute_with_retry(self, task_id: int, func: Callable, 
                           db_manager: 'TaskStateManager') -> dict:
        """带重试的任务执行"""
        retry_count = 0
        last_error = None
        
        while True:
            try:
                result = func()
                return {
                    'success': True,
                    'result': result,
                    'retry_count': retry_count
                }
            except Exception as e:
                last_error = e
                error_type = self.classify_error(e)
                
                if not self.policy.should_retry(error_type, retry_count):
                    # 不重试，记录失败
                    return {
                        'success': False,
                        'error': str(e),
                        'error_type': error_type.value,
                        'retry_count': retry_count
                    }
                
                # 准备重试
                delay = self.policy.calculate_delay(retry_count)
                self.retry_log.append({
                    'task_id': task_id,
                    'attempt': retry_count + 1,
                    'error': str(e),
                    'error_type': error_type.value,
                    'delay': round(delay, 2),
                    'timestamp': datetime.now().isoformat()
                })
                
                # 更新数据库状态为 retry_waiting
                db_manager.advance_state(
                    task_id, 
                    TaskState.RETRY_WAITING,
                    context={
                        'retry_count': retry_count + 1,
                        'error': str(e),
                        'error_type': error_type.value,
                        'next_retry_at': (datetime.now() + timedelta(seconds=delay)).isoformat()
                    }
                )
                
                # 等待重试
                time.sleep(delay)
                
                retry_count += 1
                db_manager.advance_state(
                    task_id,
                    TaskState.RETRYING,
                    context={'retry_count': retry_count}
                )
```

```python
#!/usr/bin/env python3
"""
validation_pipeline.py - OpenClaw 结果验证管道
实现 P9: Result Validation 模式
"""

import os
import json
import re
from typing import Optional
from dataclasses import dataclass, field
from datetime import datetime

@dataclass
class ValidationResult:
    """验证结果"""
    passed: bool = False
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    suggestions: list[str] = field(default_factory=list)
    checks_performed: int = 0
    checks_passed: int = 0
    
    @property
    def pass_rate(self) -> float:
        if self.checks_performed == 0:
            return 1.0
        return self.checks_passed / self.checks_performed
    
    def add_error(self, msg: str):
        self.errors.append(msg)
        self.checks_performed += 1
    
    def add_warning(self, msg: str):
        self.warnings.append(msg)
        self.checks_performed += 1
        self.checks_passed += 1  # 警告不算失败
    
    def add_pass(self, msg: str = "check passed"):
        self.checks_performed += 1
        self.checks_passed += 1
    
    def finalize(self) -> bool:
        self.passed = len(self.errors) == 0
        return self.passed


class ValidationPipeline:
    """验证管道 - 按规则链执行检查"""
    
    def __init__(self, task_id: int, output_dir: str):
        self.task_id = task_id
        self.output_dir = output_dir
        self.result = ValidationResult()
    
    def run_all(self, task_record: dict) -> ValidationResult:
        """执行完整验证流程"""
        # 1. 检查必填字段
        self._check_required_fields(task_record)
        
        # 2. 检查输出文件存在性
        self._check_output_files(task_record)
        
        # 3. 检查文件大小
        self._check_file_sizes()
        
        # 4. 检查内容完整性
        self._check_content_completeness(task_record)
        
        # 5. 检查 execution_log 长度
        self._check_execution_log(task_record)
        
        # 6. 检查 result_summary 长度
        self._check_result_summary(task_record)
        
        return self.result
    
    def _check_required_fields(self, task: dict):
        """检查任务记录中的必填字段"""
        required = ['execution_log', 'result_summary', 'task_summary']
        for field in required:
            value = task.get(field, '')
            if not value or len(str(value).strip()) == 0:
                self.result.add_error(f"缺少必填字段: {field}")
            else:
                self.result.add_pass(f"字段 {field} 存在")
    
    def _check_execution_log(self, task: dict):
        """检查 execution_log 长度 >= 200 字"""
        log_text = str(task.get('execution_log', ''))
        char_count = len(log_text)
        
        if char_count < 200:
            self.result.add_error(
                f"execution_log 仅 {char_count} 字，要求 ≥200 字"
            )
        else:
            self.result.add_pass(
                f"execution_log {char_count} 字 ✓"
            )
    
    def _check_result_summary(self, task: dict):
        """检查 result_summary 长度 >= 50 字"""
        summary = str(task.get('result_summary', ''))
        char_count = len(summary)
        
        if char_count < 50:
            self.result.add_error(
                f"result_summary 仅 {char_count} 字，要求 ≥50 字"
            )
        else:
            self.result.add_pass(
                f"result_summary {char_count} 字 ✓"
            )
    
    def _check_output_files(self, task: dict):
        """检查附件文件是否存在"""
        attachments = self._get_attachments(self.task_id)
        
        if not attachments:
            self.result.add_warning("无附件文件（如果任务不需要产出文件可忽略）")
            return
        
        for att in attachments:
            full_path = os.path.join(
                '/Users/mettlyz/.openclaw/workspace',
                att.get('url', '')
            )
            if os.path.exists(full_path):
                self.result.add_pass(f"附件存在: {att['filename']}")
            else:
                self.result.add_error(f"附件缺失: {att['filename']} (路径: {full_path})")
    
    def _check_file_sizes(self):
        """检查文件大小是否合理"""
        # 跳过，上面 _check_output_files 已经检查了
    
    def _check_content_completeness(self, task: dict):
        """根据任务类型检查内容完整性"""
        details = task.get('details', '')
        
        # 如果任务要求产出物，检查是否有相关描述
        if '产出' in details or '产出物' in details or '报告' in details:
            attachments = self._get_attachments(self.task_id)
            if not attachments:
                self.result.add_error("任务要求产出物但无附件")
            else:
                self.result.add_pass(f"产出物已生成 ({len(attachments)} 个附件)")
```

```python
#!/usr/bin/env python3
"""
plan_execute_reflect.py - OpenClaw 规划-执行-反思循环
实现 P1: Plan-Execute-Reflect 模式
"""

import json
from typing import Optional
from datetime import datetime

class TaskPlanner:
    """任务规划器"""
    
    COMPLEXITY_THRESHOLDS = {
        'simple': {'max_words': 50, 'max_steps': 2},
        'medium': {'max_words': 200, 'max_steps': 5},
        'complex': {'max_words': 500, 'max_steps': 15},
    }
    
    def assess_complexity(self, task_details: str) -> str:
        """评估任务复杂度"""
        word_count = len(task_details)
        
        if word_count <= self.COMPLEXITY_THRESHOLDS['simple']['max_words']:
            return 'simple'
        elif word_count <= self.COMPLEXITY_THRESHOLDS['medium']['max_words']:
            return 'medium'
        else:
            return 'complex'
    
    def generate_plan(self, task_id: int, task_details: str, 
                      complexity: str) -> dict:
        """生成执行计划"""
        plan = {
            'task_id': task_id,
            'complexity': complexity,
            'generated_at': datetime.now().isoformat(),
            'steps': self._generate_steps(task_details, complexity),
            'required_tools': self._identify_tools(task_details),
            'validation_criteria': self._define_validation(task_details),
            'estimated_duration': self._estimate_duration(complexity),
        }
        return plan
    
    def _generate_steps(self, details: str, complexity: str) -> list[dict]:
        """根据任务详情生成执行步骤"""
        steps = []
        
        # 通用步骤
        steps.append({
            'order': 1,
            'action': 'analyze_requirements',
            'description': '分析任务需求',
            'expected_output': '需求分析结果'
        })
        
        if complexity in ('medium', 'complex'):
            steps.append({
                'order': 2,
                'action': 'research_and_gather',
                'description': '收集必要信息和资料',
                'expected_output': '信息集合'
            })
        
        steps.append({
            'order': len(steps) + 1,
            'action': 'execute_core_task',
            'description': '执行核心任务',
            'expected_output': '任务产出物'
        })
        
        if complexity in ('medium', 'complex'):
            steps.append({
                'order': len(steps) + 1,
                'action': 'quality_check',
                'description': '质量检查和验证',
                'expected_output': '验证结果'
            })
        
        steps.append({
            'order': len(steps) + 1,
            'action': 'finalize_and_save',
            'description': '保存产出物并更新数据库',
            'expected_output': '完成状态'
        })
        
        return steps
    
    def _identify_tools(self, details: str) -> list[str]:
        """识别可能需要的工具"""
        tools = []
        
        tool_keywords = {
            'web_fetch': ['搜索', '查找', '研究', '调研', '分析', 'fetch'],
            'exec': ['运行', '执行', '脚本', '命令', '数据库'],
            'read': ['读取', '查看', '检查', '文件'],
            'write': ['写入', '创建', '保存', '生成', '文档', '报告'],
            'edit': ['编辑', '修改'],
            'memory_search': ['回忆', '记忆', '查找历史'],
            'sessions_spawn': ['子代理', 'spawn', '并行'],
        }
        
        for tool, keywords in tool_keywords.items():
            if any(kw in details for kw in keywords):
                tools.append(tool)
        
        return tools
    
    def _define_validation(self, details: str) -> list[str]:
        """定义验证标准"""
        criteria = []
        
        if '产出物' in details or '报告' in details:
            criteria.append('产出物文件已创建')
            criteria.append('产出物内容不为空')
        
        if '数据库' in details or '更新' in details:
            criteria.append('数据库记录已更新')
        
        if '执行日志' in details or 'execution_log' in details:
            criteria.append('execution_log >= 200字')
        
        if '结果摘要' in details or 'result_summary' in details:
            criteria.append('result_summary >= 50字')
        
        if not criteria:
            criteria.append('任务目标已达成')
        
        return criteria
    
    def _estimate_duration(self, complexity: str) -> str:
        """估算执行时间"""
        durations = {
            'simple': '1-2 分钟',
            'medium': '3-10 分钟',
            'complex': '10-30 分钟',
        }
        return durations.get(complexity, '未知')


class TaskReflector:
    """任务反思器"""
    
    def reflect(self, task_id: int, plan: dict, actual_result: dict) -> dict:
        """对比计划与实际结果，生成反思"""
        reflection = {
            'task_id': task_id,
            'reflected_at': datetime.now().isoformat(),
            'plan_adherence': self._calculate_adherence(plan, actual_result),
            'deviations': self._find_deviations(plan, actual_result),
            'lessons_learned': self._extract_lessons(plan, actual_result),
            'improvements': self._suggest_improvements(plan, actual_result),
        }
        return reflection
    
    def _calculate_adherence(self, plan: dict, result: dict) -> float:
        """计算计划遵从度 (0-1)"""
        planned_steps = len(plan.get('steps', []))
        completed_steps = result.get('completed_steps', planned_steps)
        
        if planned_steps == 0:
            return 1.0
        
        adherence = completed_steps / planned_steps
        # 考虑重试因素
        if result.get('retries', 0) > 0:
            adherence *= 0.9  # 每次重试扣10%遵从度
        
        return round(min(adherence, 1.0), 2)
    
    def _find_deviations(self, plan: dict, result: dict) -> list[dict]:
        """找出偏离计划的部分"""
        deviations = []
        
        planned_duration = plan.get('estimated_duration', '')
        actual_duration = result.get('actual_duration', '')
        
        if planned_duration and actual_duration:
            deviations.append({
                'type': 'duration',
                'planned': planned_duration,
                'actual': actual_duration
            })
        
        if result.get('errors'):
            deviations.append({
                'type': 'errors',
                'count': len(result['errors']),
                'details': result['errors']
            })
        
        return deviations
    
    def _extract_lessons(self, plan: dict, result: dict) -> list[str]:

        lessons = []
        
        if result.get('retries', 0) > 0:
            lessons.append(
                f"任务执行中遇到 {result['retries']} 次重试，"

            )
        
        if result.get('execution_time', 0) > 600:
            lessons.append("任务执行时间超过10分钟，建议拆分或优化")
        
        if result.get('validation_passed', False):
            lessons.append("结果验证通过，当前方法有效")
        
        return lessons
    
    def _suggest_improvements(self, plan: dict, result: dict) -> list[str]:
        """提出改进建议"""
        improvements = []
        
        if result.get('retries', 0) >= 2:
            improvements.append("考虑增加前置检查，减少重试次数")
        
        if not result.get('validation_passed', True):
            improvements.append("加强结果验证标准")
        
        return improvements
```

## 模块五: 子任务拆分框架 (P7 Subtask Decomposition)

```python
#!/usr/bin/env python3
"""
subtask_decomposer.py - OpenClaw 子任务拆分引擎
实现 P7: Subtask Decomposition 模式
"""

from typing import Optional
from dataclasses import dataclass, field
from datetime import datetime

@dataclass
class SubTask:
    """子任务定义"""
    id: str
    title: str
    description: str
    status: str = "pending"
    depends_on: list[str] = field(default_factory=list)
    output_file: Optional[str] = None
    parent_task_id: Optional[int] = None
    created_at: str = ""
    
    def __post_init__(self):
        if not self.created_at:
            self.created_at = datetime.now().isoformat()

class SubTaskGraph:
    """DAG-based 子任务图"""
    
    def __init__(self):
        self.nodes: dict[str, SubTask] = {}
        self.edges: dict[str, list[str]] = {}  # task_id -> [dependent_ids]
    
    def add_task(self, task: SubTask):
        """添加子任务"""
        self.nodes[task.id] = task
        self.edges[task.id] = []
        
        # 建立反向依赖
        for dep_id in task.depends_on:
            if dep_id in self.edges:
                self.edges[dep_id].append(task.id)
    
    def get_ready_tasks(self) -> list[SubTask]:
        """获取所有可执行的子任务（依赖已满足）"""
        ready = []
        for task_id, task in self.nodes.items():
            if task.status != "pending":
                continue
            
            # 检查所有依赖是否完成
            deps_satisfied = all(
                self.nodes[dep_id].status == "completed"
                for dep_id in task.depends_on
                if dep_id in self.nodes
            )
            
            if deps_satisfied:
                ready.append(task)
        
        return ready
    
    def mark_complete(self, task_id: str):
        """标记子任务完成"""
        if task_id in self.nodes:
            self.nodes[task_id].status = "completed"
    
    def is_all_complete(self) -> bool:
        """检查是否所有子任务都已完成"""
        return all(t.status == "completed" for t in self.nodes.values())
    
    def get_progress(self) -> dict:
        """获取整体进度"""
        total = len(self.nodes)
        completed = sum(1 for t in self.nodes.values() if t.status == "completed")
        pending = sum(1 for t in self.nodes.values() if t.status == "pending")
        executing = sum(1 for t in self.nodes.values() if t.status == "executing")
        
        return {
            'total': total,
            'completed': completed,
            'pending': pending,
            'executing': executing,
            'progress_percent': round(completed / total * 100, 1) if total > 0 else 0
        }
    
    def to_dict(self) -> dict:
        """序列化为字典"""
        return {
            'nodes': {tid: {
                'id': t.id,
                'title': t.title,
                'status': t.status,
                'depends_on': t.depends_on,
            } for tid, t in self.nodes.items()},
            'progress': self.get_progress()
        }


class SubTaskDecomposer:
    """子任务拆分器"""
    
    # 预定义的任务模板
    TEMPLATES = {
        'report_generation': {
            'description': '生成研究报告',
            'steps': [
                ('research', '信息收集与研究', []),
                ('outline', '生成报告大纲', ['research']),
                ('write_draft', '撰写报告草稿', ['outline']),
                ('review', '审核和修改', ['write_draft']),
                ('finalize', '最终定稿和保存', ['review']),
            ]
        },
        'data_analysis': {
            'description': '数据分析任务',
            'steps': [
                ('fetch_data', '获取数据', []),
                ('clean_data', '数据清洗', ['fetch_data']),
                ('analyze', '数据分析', ['clean_data']),
                ('visualize', '结果可视化', ['analyze']),
                ('report', '生成分析报告', ['visualize']),
            ]
        },
        'multi_file_output': {
            'description': '多文件产出任务',
            'steps': [
                ('gather_info', '收集信息', []),
                ('file_a', '生成文件A', ['gather_info']),
                ('file_b', '生成文件B', ['gather_info']),
                ('summary', '生成汇总文件', ['file_a', 'file_b']),
            ]
        },
    }
    
    def decompose(self, task_id: int, task_type: str, 
                  title: str, details: str) -> SubTaskGraph:
        """根据任务类型和详情拆分子任务"""
        graph = SubTaskGraph()
        
        # 查找匹配的模板
        template = self._match_template(task_type, details)
        
        if template:
            # 使用模板
            for step_id, step_title, deps in template['steps']:
                subtask = SubTask(
                    id=f"{task_id}-{step_id}",
                    title=f"[{title}] {step_title}",
                    description=details,
                    depends_on=[f"{task_id}-{d}" for d in deps],
                    parent_task_id=task_id
                )
                graph.add_task(subtask)
        else:
            # 默认拆分
            subtasks = [
                ('analyze', '分析任务需求', []),
                ('execute', '执行任务', ['analyze']),
                ('verify', '验证结果', ['execute']),
            ]
            
            for step_id, step_title, deps in subtasks:
                subtask = SubTask(
                    id=f"{task_id}-{step_id}",
                    title=f"[{title}] {step_title}",
                    description=details,
                    depends_on=[f"{task_id}-{d}" for d in deps],
                    parent_task_id=task_id
                )
                graph.add_task(subtask)
        
        return graph
    
    def _match_template(self, task_type: str, details: str) -> Optional[dict]:
        """匹配最合适的模板"""
        # 直接类型匹配
        if task_type in self.TEMPLATES:
            return self.TEMPLATES[task_type]
        
        # 关键词匹配
        details_lower = details.lower()
        if any(kw in details_lower for kw in ['报告', '研究', 'report']):
            return self.TEMPLATES.get('report_generation')
        if any(kw in details_lower for kw in ['分析', '数据', 'analysis']):
            return self.TEMPLATES.get('data_analysis')
        if any(kw in details_lower for kw in ['多个', '文件', '产出物', '附件']):
            return self.TEMPLATES.get('multi_file_output')
        
        return None
```
