```python
# mcp/protocol.py
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional
from enum import Enum
import uuid
import time

class StatusCode(Enum):
    SUCCESS = "success"
    ERROR = "error"
    TIMEOUT = "timeout"
    RATE_LIMITED = "rate_limited"

@dataclass
class MCPRequest:
    """MCP 请求协议"""
    request_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    session_id: Optional[str] = None
    skill: str = ""
    action: str = ""
    parameters: Dict[str, Any] = field(default_factory=dict)
    context: Dict[str, Any] = field(default_factory=dict)
    timestamp: float = field(default_factory=time.time)

@dataclass
class MCPResponse:
    """MCP 响应协议"""
    request_id: str
    status: StatusCode
    data: Any = None
    error: Optional[str] = None
    metadata: Dict[str, Any] = field(default_factory=dict)
    duration_ms: float = 0.0

class MCPServer(ABC):
    """MCP Server 抽象基类"""
    
    @abstractmethod
    async def call(self, request: MCPRequest) -> MCPResponse:
        pass
    
    @abstractmethod
    def get_capabilities(self) -> Dict[str, Any]:
        pass
    
    async def health_check(self) -> Dict[str, Any]:
        return {"status": "healthy", "skill": self.__class__.__name__}
```

### 1.2 MCP Server 基础类

```python
# mcp/server.py
import asyncio
import time
from typing import Dict, Any
from .protocol import MCPRequest, MCPResponse, StatusCode, MCPServer
from ..observability import MetricsCollector, TracingManager, CostEngine

class BaseMCPServer(MCPServer):
    """带可观测性的 MCP Server 基类"""
    
    def __init__(self, skill_name: str):
        self.skill_name = skill_name
        self.metrics = MetricsCollector()
        self.tracing = TracingManager(service_name=f"mcp-{skill_name}")
        self.cost_engine = CostEngine()
    
    async def call(self, request: MCPRequest) -> MCPResponse:
        start_time = time.time()
        span = self.tracing.create_span(
            f"{self.skill_name}.{request.action}",
            {"request_id": request.request_id}
        )
        
        try:
            # 实际业务逻辑
            result = await self._handle_request(request)
            
            duration = (time.time() - start_time) * 1000
            self.metrics.record_request(
                self.skill_name, request.action, "success", duration/1000
            )
            
            span.set_attribute("status", "success")
            span.end()
            
            return MCPResponse(
                request_id=request.request_id,
                status=StatusCode.SUCCESS,
                data=result,
                duration_ms=duration
            )
            
        except Exception as e:
            duration = (time.time() - start_time) * 1000
            self.metrics.record_request(
                self.skill_name, request.action, "error", duration/1000
            )
            self.metrics.error_count.labels(self.skill_name, type(e).__name__).inc()
            
            span.set_attribute("status", "error")
            span.set_attribute("error", str(e))
            span.end()
            
            return MCPResponse(
                request_id=request.request_id,
                status=StatusCode.ERROR,
                error=str(e),
                duration_ms=duration
            )
    
    @abstractmethod
    async def _handle_request(self, request: MCPRequest) -> Any:
        """子类实现具体业务逻辑"""
        pass
```

```python
# mcp/skills/memory_server.py
from typing import List, Dict, Any
from ..server import BaseMCPServer
from ..protocol import MCPRequest
from ...memory import HybridMemoryEngine

class MemoryMCPServer(BaseMCPServer):
    """Memory 检索 MCP Server"""
    
    def __init__(self):
        super().__init__("memory")
        self.engine = HybridMemoryEngine()
    
    def get_capabilities(self) -> Dict[str, Any]:
        return {
            "skill": "memory",
            "actions": ["search", "insert", "update", "delete", "recall"],
            "description": "混合记忆检索引擎，支持向量+图谱+全文检索",
            "performance": {
                "p95_latency": "100ms",
                "throughput": "100 QPS"
            }
        }
    
    async def _handle_request(self, request: MCPRequest) -> Any:
        action = request.action
        params = request.parameters
        
        if action == "search":
            return await self._search(params)
        elif action == "insert":
            return await self._insert(params)
        elif action == "update":
            return await self._update(params)
        elif action == "delete":
            return await self._delete(params)
        elif action == "recall":
            return await self._recall(params)
        else:
            raise ValueError(f"Unknown action: {action}")
    
    async def _search(self, params: Dict[str, Any]) -> Dict[str, Any]:
        query = params["query"]
        top_k = params.get("top_k", 10)
        domains = params.get("domains")
        
        result = await self.engine.search(query, top_k=top_k, domains=domains)
        
        # 记录成本
        skill_cost = self.cost_engine.calculate_skill_cost("memory_search")
        self.cost_engine.daily_usage[params.get("date", "today")]["total_cost"] += skill_cost
        
        return {
            "query": query,
            "results": [r.dict() for r in result.results],
            "context": result.context,
            "metadata": result.metadata
        }
    
    async def _insert(self, params: Dict[str, Any]) -> Dict[str, Any]:
        content = params["content"]
        metadata = params.get("metadata", {})
        
        memory_id = await self.engine.insert(content, **metadata)
        
        return {
            "memory_id": memory_id,
            "status": "inserted"
        }
    
    async def _recall(self, params: Dict[str, Any]) -> Dict[str, Any]:
        """会话上下文召回"""
        session_id = params["session_id"]
        window_size = params.get("window_size", 5)
        
        context = await self.engine.get_session_context(session_id, window_size)
        
        return {
            "session_id": session_id,
            "context_items": context
        }
```

```python
# mcp/skills/web_search_server.py
from typing import List, Dict, Any
import httpx
from ..server import BaseMCPServer
from ..protocol import MCPRequest

class WebSearchMCPServer(BaseMCPServer):
    """Web 搜索 MCP Server"""
    
    def __init__(self, api_key: str = None):
        super().__init__("web_search")
        self.api_key = api_key
        self.base_url = "https://api.tavily.com"
    
    def get_capabilities(self) -> Dict[str, Any]:
        return {
            "skill": "web_search",
            "actions": ["search", "news", "image", "extract"],
            "description": "全网搜索能力，支持网页、新闻、图片搜索",
            "providers": ["tavily", "serper", "google"]
        }
    
    async def _handle_request(self, request: MCPRequest) -> Any:
        action = request.action
        params = request.parameters
        
        if action == "search":
            return await self._search(params)
        elif action == "news":
            return await self._news(params)
        elif action == "extract":
            return await self._extract(params)
        else:
            raise ValueError(f"Unknown action: {action}")
    
    async def _search(self, params: Dict[str, Any]) -> Dict[str, Any]:
        query = params["query"]
        max_results = params.get("max_results", 5)
        search_depth = params.get("search_depth", "basic")
        
        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.post(
                f"{self.base_url}/search",
                json={
                    "api_key": self.api_key,
                    "query": query,
                    "max_results": max_results,
                    "search_depth": search_depth
                }
            )
            response.raise_for_status()
            data = response.json()
        
        # 计算搜索成本
        skill_cost = self.cost_engine.calculate_skill_cost("web_search")
        
        return {
            "query": query,
            "results": data.get("results", []),
            "follow_up_questions": data.get("follow_up_questions", []),
            "cost": skill_cost,
            "result_count": len(data.get("results", []))
        }
    
    async def _news(self, params: Dict[str, Any]) -> Dict[str, Any]:
        query = params["query"]
        days = params.get("days", 7)
        
        # 新闻搜索实现
        results = await self._search({
            "query": f"{query} when:{days}d",
            "max_results": params.get("max_results", 10)
        })
        
        return {
            "query": query,
            "time_window": f"{days}d",
            "results": results["results"]
        }
    
    async def _extract(self, params: Dict[str, Any]) -> Dict[str, Any]:
        """网页内容提取"""
        url = params["url"]
        
        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.post(
                f"{self.base_url}/extract",
                json={
                    "api_key": self.api_key,
                    "urls": [url]
                }
            )
            response.raise_for_status()
            data = response.json()
        
        return {
            "url": url,
            "content": data.get("results", [{}])[0].get("raw_content", ""),
            "title": data.get("results", [{}])[0].get("title", "")
        }
```

```python
# mcp/skills/file_server.py
from typing import List, Dict, Any
from pathlib import Path
import mimetypes
from ..server import BaseMCPServer
from ..protocol import MCPRequest

class FileSystemMCPServer(BaseMCPServer):
    """文件系统 MCP Server"""
    
    def __init__(self, base_path: str = "/Users/mettlyz/.openclaw/workspace"):
        super().__init__("file_system")
        self.base_path = Path(base_path).resolve()
    
    def get_capabilities(self) -> Dict[str, Any]:
        return {
            "skill": "file_system",
            "actions": ["read", "write", "list", "stat", "search", "mkdir"],
            "description": "安全的文件系统访问能力",
            "sandboxed": True,
            "base_path": str(self.base_path)
        }
    
    async def _handle_request(self, request: MCPRequest) -> Any:
        action = request.action
        params = request.parameters
        
        if action == "read":
            return await self._read(params)
        elif action == "write":
            return await self._write(params)
        elif action == "list":
            return await self._list(params)
        elif action == "stat":
            return await self._stat(params)
        elif action == "search":
            return await self._search(params)
        elif action == "mkdir":
            return await self._mkdir(params)
        else:
            raise ValueError(f"Unknown action: {action}")
    
    def _resolve_path(self, path: str) -> Path:
        """安全路径解析，防止路径遍历"""
        target = (self.base_path / path).resolve()
        if not str(target).startswith(str(self.base_path)):
            raise PermissionError(f"Access denied: {path}")
        return target
    
    async def _read(self, params: Dict[str, Any]) -> Dict[str, Any]:
        path = self._resolve_path(params["path"])
        max_size = params.get("max_size", 10 * 1024 * 1024)  # 10MB
        
        if not path.exists():
            raise FileNotFoundError(f"File not found: {path}")
        
        if path.stat().st_size > max_size:
            raise ValueError(f"File too large: {path.stat().st_size} bytes")
        
        content = path.read_text(encoding='utf-8', errors='replace')
        
        # 计算文件操作成本
        file_size = path.stat().st_size
        token_estimate = len(content) / 4  # 粗略估算
        skill_cost = self.cost_engine.calculate_skill_cost("file_read")
        
        return {
            "path": str(path.relative_to(self.base_path)),
            "content": content,
            "size_bytes": file_size,
            "encoding": "utf-8",
            "cost": skill_cost
        }
    
    async def _write(self, params: Dict[str, Any]) -> Dict[str, Any]:
        path = self._resolve_path(params["path"])
        content = params["content"]
        append = params.get("append", False)
        
        path.parent.mkdir(parents=True, exist_ok=True)
        
        if append and path.exists():
            existing = path.read_text(encoding='utf-8', errors='replace')
            content = existing + content
        
        path.write_text(content, encoding='utf-8')
        
        return {
            "path": str(path.relative_to(self.base_path)),
            "size_bytes": len(content),
            "status": "written"
        }
    
    async def _list(self, params: Dict[str, Any]) -> Dict[str, Any]:
        path = self._resolve_path(params.get("path", "."))
        recursive = params.get("recursive", False)
        
        if not path.is_dir():
            raise NotADirectoryError(f"Not a directory: {path}")
        
        files = []
        glob_pattern = "**/*" if recursive else "*"
        
        for item in path.glob(glob_pattern):
            if item.name.startswith('.'):
                continue
            rel_path = item.relative_to(self.base_path)
            files.append({
                "path": str(rel_path),
                "name": item.name,
                "type": "directory" if item.is_dir() else "file",
                "size": item.stat().st_size if item.is_file() else 0,
                "extension": item.suffix[1:] if item.suffix else ""
            })
        
        return {
            "path": str(path.relative_to(self.base_path)),
            "files": files,
            "count": len(files)
        }
    
    async def _search(self, params: Dict[str, Any]) -> Dict[str, Any]:
        pattern = params["pattern"]
        path = self._resolve_path(params.get("path", "."))
        
        matching_files = []
        for item in path.rglob(pattern):
            if item.is_file():
                rel_path = item.relative_to(self.base_path)
                matching_files.append({
                    "path": str(rel_path),
                    "size": item.stat().st_size
                })
        
        return {
            "pattern": pattern,
            "matches": matching_files,
            "count": len(matching_files)
        }
```

## 五、MCP Gateway 网关实现

```python
# mcp/gateway.py
from typing import Dict, List
import asyncio
from fastapi import FastAPI, HTTPException
from .protocol import MCPRequest, MCPResponse
from .skills.memory_server import MemoryMCPServer
from .skills.web_search_server import WebSearchMCPServer
from .skills.file_server import FileSystemMCPServer

app = FastAPI(title="MCP Gateway", version="1.0")

class MCPGateway:
    def __init__(self):
        self.servers: Dict[str, BaseMCPServer] = {}
        self._register_servers()
    
    def _register_servers(self):
        """注册所有 MCP Server"""
        self.servers["memory"] = MemoryMCPServer()
        self.servers["web_search"] = WebSearchMCPServer()
        self.servers["file_system"] = FileSystemMCPServer()
    
    async def route_request(self, request: MCPRequest) -> MCPResponse:
        """路由请求到对应的 MCP Server"""
        if request.skill not in self.servers:
            return MCPResponse(
                request_id=request.request_id,
                status=StatusCode.ERROR,
                error=f"Unknown skill: {request.skill}"
            )
        
        server = self.servers[request.skill]
        return await server.call(request)

# FastAPI 路由
gateway = MCPGateway()

@app.post("/mcp/call")
async def mcp_call(request: MCPRequest):
    response = await gateway.route_request(request)
    return response

@app.get("/mcp/capabilities")
async def get_capabilities():
    return {
        skill: server.get_capabilities()
        for skill, server in gateway.servers.items()
    }

@app.get("/health")
async def health_check():
    health_status = {}
    for skill, server in gateway.servers.items():
        health_status[skill] = await server.health_check()
    return health_status
```
