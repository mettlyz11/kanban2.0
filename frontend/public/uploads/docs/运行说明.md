#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Sobereva博客文章爬取工具
功能：爬取博客文章列表并存储到SQLite数据库
作者：自动化任务执行团队
版本：1.0.0
"""

import os
import sys
import time
import logging
import sqlite3
from datetime import datetime
from typing import List, Dict, Optional, Tuple

import requests
from bs4 import BeautifulSoup
from lxml import etree

# ==================== 配置区域 ====================
CONFIG = {
    "BASE_URL": "http://sobereva.com",
    "USER_AGENT": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
                  "(KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
    "REQUEST_TIMEOUT": 15,           # 请求超时时间（秒）
    "REQUEST_DELAY": 1.5,            # 请求间隔（秒），避免被反爬
    "MAX_RETRIES": 3,                # 最大重试次数
    "DB_FILE": "sobereva_articles.db",
    "LOG_FILE": "crawler.log",
    "LOG_LEVEL": "INFO",             # 可选: DEBUG, INFO, WARNING, ERROR
    "MAX_PAGES": 50,                 # 最大爬取页数（0表示无限制）
}
# ==================================================

# 设置日志
logging.basicConfig(
    level=getattr(logging, CONFIG["LOG_LEVEL"]),
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[
        logging.FileHandler(CONFIG["LOG_FILE"], encoding="utf-8"),
        logging.StreamHandler(sys.stdout)
    ]
)
logger = logging.getLogger(__name__)


class DatabaseManager:
    """数据库管理类，负责创建表和插入数据"""
    
    def __init__(self, db_path: str):
        self.db_path = db_path
        self.conn = None
        self.cursor = None
        self._init_database()
    
    def _init_database(self):
        """初始化数据库连接并创建表"""
        self.conn = sqlite3.connect(self.db_path)
        self.cursor = self.conn.cursor()
        
        # 创建文章表
        self.cursor.execute("""
            CREATE TABLE IF NOT EXISTS articles (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                url TEXT NOT NULL UNIQUE,
                publish_date TEXT,
                summary TEXT,
                crawl_time TEXT NOT NULL,
                page_num INTEGER DEFAULT 0
            )
        """)
        
        # 创建爬取记录表
        self.cursor.execute("""
            CREATE TABLE IF NOT EXISTS crawl_records (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                start_time TEXT NOT NULL,
                end_time TEXT,
                pages_crawled INTEGER DEFAULT 0,
                articles_found INTEGER DEFAULT 0,
                articles_new INTEGER DEFAULT 0,
                status TEXT DEFAULT 'running'
            )
        """)
        
        self.conn.commit()
        logger.info(f"数据库初始化完成: {db_path}")
    
    def url_exists(self, url: str) -> bool:
        """检查URL是否已存在"""
        self.cursor.execute("SELECT COUNT(*) FROM articles WHERE url = ?", (url,))
        return self.cursor.fetchone()[0] > 0
    
    def insert_article(self, article: Dict) -> bool:
        """插入一篇文章，返回是否为新插入"""
        if self.url_exists(article["url"]):
            return False
        
        try:
            self.cursor.execute("""
                INSERT INTO articles (title, url, publish_date, summary, crawl_time, page_num)
                VALUES (?, ?, ?, ?, ?, ?)
            """, (
                article["title"],
                article["url"],
                article.get("publish_date", ""),
                article.get("summary", ""),
                datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                article.get("page_num", 0)
            ))
            self.conn.commit()
            return True
        except sqlite3.IntegrityError as e:
            logger.warning(f"插入重复URL: {article['url']} - {e}")
            return False
    
    def insert_crawl_record(self, start_time: str) -> int:
        """插入爬取记录，返回记录ID"""
        self.cursor.execute("""
            INSERT INTO crawl_records (start_time) VALUES (?)
        """, (start_time,))
        self.conn.commit()
        return self.cursor.lastrowid
    
    def update_crawl_record(self, record_id: int, end_time: str, 
                           pages: int, found: int, new: int):
        """更新爬取记录"""
        self.cursor.execute("""
            UPDATE crawl_records 
            SET end_time=?, pages_crawled=?, articles_found=?, 
                articles_new=?, status='completed'
            WHERE id=?
        """, (end_time, pages, found, new, record_id))
        self.conn.commit()
    
    def get_article_count(self) -> int:
        """获取文章总数"""
        self.cursor.execute("SELECT COUNT(*) FROM articles")
        return self.cursor.fetchone()[0]
    
    def get_sample_articles(self, limit: int = 10) -> List[Dict]:
        """获取示例文章"""
        self.cursor.execute("""
            SELECT title, url, publish_date, summary, crawl_time 
            FROM articles ORDER BY crawl_time DESC LIMIT ?
        """, (limit,))
        rows = self.cursor.fetchall()
        return [
            {
                "title": row[0],
                "url": row[1],
                "publish_date": row[2],
                "summary": row[3],
                "crawl_time": row[4]
            }
            for row in rows
        ]
    
    def close(self):
        """关闭数据库连接"""
        if self.conn:
            self.conn.close()
            logger.info("数据库连接已关闭")


class SoberevaCrawler:
    """Sobereva博客爬虫类"""
    
    def __init__(self, config: Dict):
        self.config = config
        self.session = requests.Session()
        self.session.headers.update({"User-Agent": config["USER_AGENT"]})
        self.db = DatabaseManager(config["DB_FILE"])
        self.total_found = 0
        self.total_new = 0
        self.pages_crawled = 0
    
    def fetch_page(self, url: str) -> Optional[str]:
        """获取网页内容，支持重试"""
        for attempt in range(1, self.config["MAX_RETRIES"] + 1):
            try:
                logger.debug(f"正在请求: {url} (尝试 {attempt}/{self.config['MAX_RETRIES']})")
                response = self.session.get(
                    url, 
                    timeout=self.config["REQUEST_TIMEOUT"]
                )
                response.encoding = "utf-8"
                
                if response.status_code == 200:
                    return response.text
                elif response.status_code == 404:
                    logger.warning(f"页面不存在: {url}")
                    return None
                else:
                    logger.warning(f"HTTP {response.status_code}: {url}")
                    
            except requests.RequestException as e:
                logger.error(f"请求失败: {url} - {e}")
                if attempt < self.config["MAX_RETRIES"]:
                    wait_time = attempt * 2
                    logger.info(f"等待 {wait_time} 秒后重试...")
                    time.sleep(wait_time)
                else:
                    logger.error(f"达到最大重试次数，放弃: {url}")
                    return None
        
        return None
    
    def parse_article_list(self, html: str, page_num: int) -> List[Dict]:
        """解析文章列表页"""
        articles = []
        soup = BeautifulSoup(html, "lxml")
        
        # 常见博客文章列表选择器（根据实际页面结构调整）
        # 方法1: 查找所有文章条目（适用于大多数WordPress主题）
        article_items = soup.select("article, .post, .entry, .hentry, li[class*='post']")
        
        if not article_items:
            # 方法2: 查找所有链接包含日期的元素
            article_items = soup.find_all("div", class_=lambda x: x and "post" in x.lower())
        
        if not article_items:
            # 方法3: 直接查找所有h2/h3标题下的链接
            for header in soup.find_all(["h2", "h3"]):
                link = header.find("a")
                if link and link.get("href"):
                    articles.append({
                        "title": link.get_text(strip=True),
                        "url": self._make_absolute_url(link["href"]),
                        "publish_date": "",
                        "summary": "",
                        "page_num": page_num
                    })
            return articles
        
        for item in article_items:
            # 提取标题和链接
            title_elem = item.find("a")
            if not title_elem:
                continue
            
            title = title_elem.get_text(strip=True)
            if not title:
                continue
            
            url = self._make_absolute_url(title_elem.get("href", ""))
            
            # 提取发布日期
            date_elem = item.find("time") or item.find(class_=lambda x: x and "date" in x.lower())
            publish_date = ""
            if date_elem:
                publish_date = date_elem.get_text(strip=True)
                # 尝试获取datetime属性
                if date_elem.get("datetime"):
                    publish_date = date_elem["datetime"]
            
            # 提取摘要
            summary_elem = item.find(class_=lambda x: x and "summary" in x.lower()) or \
                          item.find(class_="entry-content") or \
                          item.find("p")
            summary = ""
            if summary_elem:
                summary = summary_elem.get_text(strip=True)[:200]  # 截取前200字符
            
            articles.append({
                "title": title,
                "url": url,
                "publish_date": publish_date,
                "summary": summary,
                "page_num": page_num
            })
        
        return articles
    
    def _make_absolute_url(self, url: str) -> str:
        """将相对URL转换为绝对URL"""
        if url.startswith("http"):
            return url
        if url.startswith("//"):
            return "https:" + url
        if url.startswith("/"):
            return self.config["BASE_URL"] + url
        return self.config["BASE_URL"] + "/" + url
    
    def get_next_page_url(self, soup: BeautifulSoup, current_url: str) -> Optional[str]:
        """获取下一页URL"""
        # 方法1: 查找带有"下一页"或"Next"的链接
        next_links = soup.select("a.next, a[rel='next'], a:contains('下一页'), a:contains('Next')")
        for link in next_links:
            href = link.get("href")
            if href:
                return self._make_absolute_url(href)
        
        # 方法2: 查找分页数字链接
        pagination = soup.find(class_=lambda x: x and "pagination" in x.lower())
        if pagination:
            current_page = None
            for a in pagination.find_all("a"):
                if a.get("class") and "current" in a.get("class", []):
                    current_page = int(a.get_text(strip=True))
                    break
            
            if current_page:
                next_page = current_page + 1
                for a in pagination.find_all("a"):
                    try:
                        page_num = int(a.get_text(strip=True))
                        if page_num == next_page:
                            return self._make_absolute_url(a["href"])
                    except ValueError:
                        continue
        
        # 方法3: 基于URL模式生成下一页
        import re
        match = re.search(r'/page/(\d+)/', current_url)
        if match:
            current_page = int(match.group(1))
            next_url = current_url.replace(f"/page/{current_page}/", f"/page/{current_page+1}/")
            return next_url
        
        # 方法4: 尝试从URL参数中获取
        if "?page=" in current_url:
            import urllib.parse
            parsed = urllib.parse.urlparse(current_url)
            params = dict(urllib.parse.parse_qsl(parsed.query))
            if "page" in params:
                params["page"] = str(int(params["page"]) + 1)
                new_query = urllib.parse.urlencode(params)
                return urllib.parse.urlunparse(parsed._replace(query=new_query))
        
        return None
    
    def crawl(self):
        """执行爬取主流程"""
        start_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        record_id = self.db.insert_crawl_record(start_time)
        
        logger.info("=" * 60)
        logger.info("Sobereva博客爬取工具 v1.0 启动")
        logger.info(f"目标网站: {self.config['BASE_URL']}")
        logger.info(f"数据库文件: {self.config['DB_FILE']}")
        logger.info(f"开始时间: {start_time}")
        logger.info("=" * 60)
        
        current_url = self.config["BASE_URL"]
        max_pages = self.config["MAX_PAGES"]
        page_count = 0
        
        while current_url and (max_pages == 0 or page_count < max_pages):
            page_count += 1
            logger.info(f"\n--- 正在爬取第 {page_count} 页 ---")
            logger.info(f"URL: {current_url}")
            
            html = self.fetch_page(current_url)
            if not html:
                logger.warning(f"获取页面失败，停止爬取")
                break
            
            articles = self.parse_article_list(html, page_count)
            logger.info(f"本页发现 {len(articles)} 篇文章")
            
            new_count = 0
            for article in articles:
                if self.db.insert_article(article):
                    new_count += 1
            
            self.total_found += len(articles
