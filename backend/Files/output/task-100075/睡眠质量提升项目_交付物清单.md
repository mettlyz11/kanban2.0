```python
import numpy as np
import onnxruntime as ort

class SleepStageInferencer:
    def __init__(self, model_path: str):
        self.session = ort.InferenceSession(model_path)
        self.input_name = self.session.get_inputs()[0].name
        self.output_name = self.session.get_outputs()[0].name
        self.labels = ["清醒", "浅睡", "深睡", "快速眼动"]

    def preprocess(self, ppg_data: np.ndarray, acc_data: np.ndarray) -> np.ndarray:
        # 标准化数据
        ppg_norm = (ppg_data - ppg_data.mean()) / (ppg_data.std() + 1e-8)
        acc_norm = (acc_data - acc_data.mean()) / (acc_data.std() + 1e-8)
        # 拼接数据，形状为 (1, 2, 750) 对应30秒25Hz采样
        input_data = np.stack([ppg_norm, acc_norm], axis=0)[np.newaxis, :].astype(np.float32)
        return input_data

    def infer(self, ppg_data: np.ndarray, acc_data: np.ndarray) -> str:
        input_data = self.preprocess(ppg_data, acc_data)
        output = self.session.run([self.output_name], {self.input_name: input_data})[0]
        pred_idx = np.argmax(output, axis=1)[0]
        return self.labels[pred_idx]
```
