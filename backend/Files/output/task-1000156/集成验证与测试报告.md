### 1.2 模拟负载构造方案

为了真实模拟用户请求的“正常-高峰-恢复”场景，采用基于时间的流量生成器。该生成器使用 `asyncio` 异步调度，通过调整每秒钟的请求数量（RPS）和 prompt 长度分布来模拟不同阶段。

#### 1.2.1 负载阶段定义

| 阶段 | 持续时间 | RPS | 平均 prompt 长度 (tokens) | 说明 |
|------|----------|-----|--------------------------|------|
| 平稳期 | 0s – 120s | 50 ± 5 | 1024 | 系统正常运行 |
| 爬升期 | 120s – 180s | 50 → 200 (线性) | 1024 | 请求量逐步增加 |
| 高峰期 | 180s – 360s | 200 ± 10 | 1024 | 持续高负载 |
| 恢复期 | 360s – 420s | 200 → 50 (线性) | 1024 | 请求量逐步回落 |
| 稳定期 | 420s – 540s | 50 ± 5 | 1024 | 系统恢复后观察 |

#### 1.2.2 负载构造代码（完整可运行）

```python
# load_generator.py
import asyncio
import random
import time
import json
from datetime import datetime

class LoadGenerator:
    def __init__(self, target_url: str, scheduler):
        self.target = target_url
        self.scheduler = scheduler  # 自适应模块调度器
        self.running = False

    async def simulate_load(self, phase_def: dict):
        """根据阶段定义发送请求"""
        start_time = time.time()
        duration = phase_def['duration']
        base_rps = phase_def['rps']
        prompt_len = phase_def['prompt_len']

        while (time.time() - start_time) < duration:
            # 对于爬升/恢复期，rps 随时间线性变化
            elapsed = time.time() - start_time
            if 'rps_end' in phase_def:
                rps = base_rps + (phase_def['rps_end'] - base_rps) * (elapsed / duration)
            else:
                rps = base_rps

            # 加入随机扰动
            rps = max(1, int(rps + random.gauss(0, 0.05 * rps)))

            # 生成并发送本秒内的请求
            tasks = []
            for _ in range(rps):
                prompt_length = max(64, int(prompt_len + random.gauss(0, 0.1 * prompt_len)))
                tasks.append(self._send_request(prompt_length))
            await asyncio.gather(*tasks, return_exceptions=True)

            # 等待剩余时间，确保每秒精准
            await asyncio.sleep(1.0 - (time.time() % 1.0))

    async def _send_request(self, prompt_len: int):
        """模拟一次外部请求（实际使用 aiohttp 可替换）"""
        async with self.scheduler.acquire():
            # 模拟处理延迟（受 prompt 长度影响）
            latency = 0.001 * prompt_len + random.uniform(0.005, 0.020)
            await asyncio.sleep(latency)
            # 记录请求信息到监控组件
            self.scheduler.record_request(prompt_len, latency)

# 使用示例：
# scheduler = AdaptiveScheduler(...)
# gen = LoadGenerator("http://api:8080", scheduler)
# asyncio.run(gen.simulate_load({'duration': 60, 'rps': 50, 'prompt_len': 1024}))
```

```python
# test_pid_controller.py
import pytest
from adaptive.pid import PIDController

def test_pid_response():
    pid = PIDController(Kp=1.0, Ki=0.1, Kd=0.05, setpoint=0.2, output_limits=(-1, 1))
    # 模拟一个从 0.15 到 0.2 的步骤变化
    control = pid.update(0.15, dt=0.1)
    assert abs(control - 0.05) < 1e-3

    # 第二次更新
    control2 = pid.update(0.19, dt=0.1)
    assert control2 < 0.01  # 误差变小时控制量趋近0

def test_integral_windup():
    pid = PIDController(Kp=0.5, Ki=0.2, Kd=0, setpoint=0.0, output_limits=(-1, 1))
    # 持续大误差，积分累积
    for _ in range(100):
        pid.update(1.0, dt=0.1)
    assert pid.integral <= 1.0 / 0.2  # 积分项限幅后不应无限增长
```

```python
from adaptive.monitor import RequestMonitor

def test_monitor_aggregation():
```

### 3.1 测试目标
