# Agent 可观测性体系设计 v1.0

## 一、体系概览

### 1.1 可观测性三大支柱
1. **Metrics (指标)** - 聚合统计，告警触发
2. **Traces (链路)** - 请求全链路追踪
3. **Logs (日志)** - 详细上下文日志

### 1.2 设计原则
- **零侵入**：SDK 自动埋点，业务代码无感知
- **低开销**：采集开销 < 5% CPU
- **高采样**：错误全采样，正常流量采样 10%
- **实时性**：告警延迟 < 30s

## 二、Metrics 指标体系

### 2.2 指标采集实现

```python
# metrics_collector.py
from prometheus_client import Histogram, Counter, Gauge

class MetricsCollector:
    def __init__(self):
        self.request_duration = Histogram(
            'mcp_request_duration_seconds',
            'MCP request duration',
            ['skill', 'action', 'status']
        )
        self.token_usage = Counter(
            'llm_token_usage_total',
            'Total LLM token usage',
            ['model', 'skill', 'user_id']
        )
        self.error_count = Counter(
            'mcp_request_errors_total',
            'Total MCP request errors',
            ['skill', 'error_type']
        )
    
    def record_request(self, skill, action, status, duration, tokens=0):
        self.request_duration.labels(skill, action, status).observe(duration)
        if tokens > 0:
            self.token_usage.labels("default", skill, "unknown").inc(tokens)
```

### 3.2 追踪实现

```python
# tracing.py
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.jaeger.thrift import JaegerExporter

class TracingManager:
    def __init__(self, service_name="mcp-server"):
        self.service_name = service_name
        self._init_tracing()
    
    def _init_tracing(self):
        jaeger_exporter = JaegerExporter(
            agent_host_name='localhost',
            agent_port=6831,
        )
        provider = TracerProvider()
        processor = BatchSpanProcessor(jaeger_exporter)
        provider.add_span_processor(processor)
        trace.set_tracer_provider(provider)
    
    def create_span(self, name, attributes=None):
        tracer = trace.get_tracer(self.service_name)
        span = tracer.start_span(name)
        if attributes:
            for k, v in attributes.items():
                span.set_attribute(k, v)
        return span
```

## 四、Cost 成本核算体系

### 4.1 计费模型
```
总成本 = Σ (技能调用费用 + LLM Token费用 + 基础设施费用)

技能调用费用 = 基础费用 * 调用次数
LLM Token费用 = (输入Token * 输入单价) + (输出Token * 输出单价)
基础设施费用 = CPU * 时长单价 + 内存 * 时长单价
```

### 4.2 成本核算实现

```python
# cost_engine.py
from datetime import datetime, timedelta
from collections import defaultdict

class CostEngine:
    PRICING = {
        'gpt-4': {'input': 0.01, 'output': 0.03},
        'kimi-k2.5': {'input': 0.002, 'output': 0.006},
        'deepseek': {'input': 0.001, 'output': 0.002},
    }
    
    SKILL_BASE_FEE = {
        'memory_search': 0.001,
        'web_search': 0.005,
        'file_read': 0.0005,
    }
    
    def __init__(self):
        self.daily_usage = defaultdict(lambda: {
            'tokens': {'input': 0, 'output': 0},
            'skill_calls': defaultdict(int),
            'total_cost': 0.0
        })
    
    def calculate_llm_cost(self, model, input_tokens, output_tokens):
        pricing = self.PRICING.get(model, self.PRICING['deepseek'])
        cost = (input_tokens * pricing['input'] + 
                output_tokens * pricing['output']) / 1000
        return cost
    
    def calculate_skill_cost(self, skill, call_count=1):
        base_fee = self.SKILL_BASE_FEE.get(skill, 0.001)
        return base_fee * call_count
    
    def get_daily_report(self, date=None):
        if date is None:
            date = datetime.now().strftime('%Y-%m-%d')
        return self.daily_usage.get(date, {
            'tokens': {'input': 0, 'output': 0},
            'skill_calls': {},
            'total_cost': 0.0
        })
```

### 4.3 预算控制机制
```python
class BudgetController:
    def __init__(self, daily_budget=50.0):
        self.daily_budget = daily_budget
        self.cost_engine = CostEngine()
        self.alert_thresholds = [0.5, 0.8, 0.95]  # 50%, 80%, 95%
    
    def check_budget_status(self, user_id=None):
        current_cost = self.cost_engine.get_daily_report()['total_cost']
        usage_ratio = current_cost / self.daily_budget
        
        alerts = []
        for threshold in self.alert_thresholds:
            if usage_ratio >= threshold and usage_ratio < threshold + 0.1:
                alerts.append(f"预算已使用 {usage_ratio*100:.1f}%")
        
        return {
            'current_cost': current_cost,
            'budget': self.daily_budget,
            'usage_ratio': usage_ratio,
            'alerts': alerts,
            'should_block': usage_ratio >= 1.0
        }
```
