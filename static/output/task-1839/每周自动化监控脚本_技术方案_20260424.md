# 和光智成竞品自动化监控脚本方案（v1.0）

## 一、整体架构设计

### 1.1 系统架构图
```
定时调度器 (Cron)
    ↓
数据采集层 (Scrapers) → 网页爬虫、RSS订阅、API调用、GitHub监控
    ↓
数据处理层 (Processors) → 数据清洗、NLP摘要、关键词提取、相似度计算
    ↓
存储层 (Storage) → SQLite/MySQL数据库、文件系统（Markdown报告）
    ↓
通知层 (Notifiers) → 邮件、企业微信、Slack告警
    ↓
可视化层 (Dashboard) → 月度报告生成、趋势分析
```

### 1.2 技术栈选型
| 模块 | 技术选型 | 说明 |
|------|---------|------|
| 爬虫框架 | Scrapy + Playwright | 处理动态网页，模拟浏览器行为 |
| RSS解析 | feedparser | 轻量级，支持标准RSS/Atom |
| NLP处理 | Transformers + Jieba | 中文文本摘要、关键词提取 |
| 调度器 | APScheduler / Cron | 定时任务调度 |
| 数据存储 | SQLite（初期） | 轻量，无需额外部署 |
| 通知推送 | requests（Webhook） | 企业微信、Slack机器人 |
| PDF解析 | PyMuPDF (fitz) | 处理白皮书、技术文档 |

### 2.2 主调度脚本 (monitor_main.py)
```python
#!/usr/bin/env python3
"""
竞品监控主调度脚本
每周一凌晨2点自动执行
"""
import yaml
import logging
from datetime import datetime
from apscheduler.schedulers.blocking import BlockingScheduler

from scrapers.web_scraper import WebScraper
from scrapers.github_scraper import GitHubScraper
from scrapers.rss_scraper import RSSScraper
from processors.content_analyzer import ContentAnalyzer
from notifiers.wecom_notifier import WecomNotifier
from reporters.weekly_report import WeeklyReporter

logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('logs/monitor.log'),
        logging.StreamHandler()
    ]
)

def load_config():
    with open('config.yaml', 'r', encoding='utf-8') as f:
        return yaml.safe_load(f)

def run_monitoring_cycle():
    """执行一次完整的监控周期"""
    config = load_config()
    logger = logging.getLogger(__name__)
    logger.info("=" * 50)
    logger.info(f"开始执行竞品监控周期: {datetime.now()}")
    
    # 1. 数据采集阶段
    all_updates = []
    
    for competitor in config['competitors']:
        logger.info(f"正在监控: {competitor['name']}")
        
        # 网页爬虫
        scraper = WebScraper(competitor)
        web_updates = scraper.run()
        all_updates.extend(web_updates)
        
        # GitHub监控
        if 'github' in competitor['urls']:
            gh_scraper = GitHubScraper(competitor['urls']['github'])
            gh_updates = gh_scraper.get_recent_activity()
            all_updates.extend(gh_updates)
        
        # RSS订阅
        if competitor['rss_feeds']:
            rss_scraper = RSSScraper(competitor['rss_feeds'])
            rss_updates = rss_scraper.fetch_new()
            all_updates.extend(rss_updates)
    
    # 2. 内容分析与去重
    analyzer = ContentAnalyzer()
    analyzed_updates = analyzer.process(all_updates)
    
    # 3. 生成周报
    reporter = WeeklyReporter()
    report_path = reporter.generate(analyzed_updates)
    
    # 4. 发送通知
    notifier = WecomNotifier(config['notifications']['wecom_webhook'])
    notifier.send_summary(analyzed_updates, report_path)
    
    logger.info(f"监控周期完成，共发现 {len(analyzed_updates)} 条重要更新")
    logger.info("=" * 50)

if __name__ == "__main__":
    scheduler = BlockingScheduler()
    scheduler.add_job(
        run_monitoring_cycle,
        'cron',
        day_of_week='mon',
        hour=2,
        minute=0
    )
    print("竞品监控服务已启动，每周一凌晨2点执行...")
    try:
        scheduler.start()
    except KeyboardInterrupt:
        print("监控服务已停止")
```

### 2.3 网页爬虫模块 (web_scraper.py)
```python
import hashlib
import requests
from bs4 import BeautifulSoup
from playwright.sync_api import sync_playwright
from datetime import datetime

class WebScraper:
    def __init__(self, competitor_config):
        self.config = competitor_config
        self.name = competitor_config['name']
    
    def get_page_hash(self, url):
        """计算页面内容hash，用于检测变化"""
        try:
            response = requests.get(url, timeout=30)
            response.raise_for_status()
            return hashlib.md5(response.content).hexdigest()
        except Exception as e:
            logging.error(f"获取页面失败 {url}: {e}")
            return None
    
    def extract_content(self, html):
        """提取页面主要内容"""
        soup = BeautifulSoup(html, 'html.parser')
        # 移除脚本和样式
        for script in soup(["script", "style"]):
            script.decompose()
        return soup.get_text(strip=True)
    
    def run(self):
        """执行爬虫任务"""
        updates = []
        
        for page_type, url in self.config['urls'].items():
            if page_type == 'github':
                continue
                
            current_hash = self.get_page_hash(url)
            
            # 与上次hash对比（从数据库读取）
            last_hash = self.get_last_hash(url)
            
            if current_hash and current_hash != last_hash:
                # 页面有更新
                with sync_playwright() as p:
                    browser = p.chromium.launch()
                    page = browser.new_page()
                    page.goto(url)
                    content = self.extract_content(page.content())
                    browser.close()
                
                updates.append({
                    'competitor': self.name,
                    'type': page_type,
                    'url': url,
                    'content': content,
                    'timestamp': datetime.now().isoformat(),
                    'importance': self._calculate_importance(content)
                })
                
                # 更新hash记录
                self.save_hash(url, current_hash)
        
        return updates
```

### 2.4 GitHub监控模块 (github_scraper.py)
```python
import requests
from datetime import datetime, timedelta

class GitHubScraper:
    def __init__(self, org_url):
        self.org_name = org_url.split('/')[-1]
        self.api_base = "https://api.github.com"
    
    def get_recent_activity(self, days=7):
        """获取组织最近7天的活动"""
        activities = []
        
        # 获取所有仓库
        repos_url = f"{self.api_base}/orgs/{self.org_name}/repos"
        repos = requests.get(repos_url).json()
        
        for repo in repos[:10]:  # 监控前10个仓库
            repo_name = repo['name']
            
            # 获取最近的commits
            commits_url = f"{self.api_base}/repos/{self.org_name}/{repo_name}/commits"
            params = {'since': (datetime.now() - timedelta(days=days)).isoformat()}
            commits = requests.get(commits_url, params=params).json()
            
            if commits:
                activities.append({
                    'competitor': self.org_name,
                    'type': 'github_commit',
                    'repo': repo_name,
                    'commits_count': len(commits),
                    'latest_commit': commits[0]['commit']['message'],
                    'url': repo['html_url'],
                    'timestamp': datetime.now().isoformat()
                })
            
            # 获取Releases
            releases_url = f"{self.api_base}/repos/{self.org_name}/{repo_name}/releases"
            releases = requests.get(releases_url).json()
            if releases:
                latest_release = releases[0]
                release_date = datetime.fromisoformat(
                    latest_release['published_at'].replace('Z', '+00:00')
                )
                if release_date > datetime.now(datetime.timezone.utc) - timedelta(days=days):
                    activities.append({
                        'competitor': self.org_name,
                        'type': 'github_release',
                        'repo': repo_name,
                        'version': latest_release['tag_name'],
                        'notes': latest_release['body'],
                        'url': latest_release['html_url'],
                        'timestamp': release_date.isoformat()
                    })
        
        return activities
```

### 2.5 内容分析模块 (content_analyzer.py)
```python
import jieba
import jieba.analyse
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

class ContentAnalyzer:
    def __init__(self):
        jieba.initialize()
        # 自定义词典（行业术语）
        self.custom_keywords = {
            '蛋白质结构预测': 10,
            '分子动力学': 8,
            'AI药物发现': 8,
            '量子计算': 7,
            '大模型': 6,
            '融资': 9,
            '战略合作': 9,
            '新品发布': 8,
            '价格调整': 7,
            '开源': 8
        }
    
    def extract_keywords(self, text, topK=10):
        """提取关键词"""
        keywords = jieba.analyse.extract_tags(text, topK=topK, withWeight=True)
        return keywords
    
    def calculate_importance(self, item):
        """计算重要性评分（0-100）"""
        score = 0
        text = item.get('content', '') + item.get('notes', '')
        
        # 基于关键词匹配
        for keyword, weight in self.custom_keywords.items():
            if keyword in text:
                score += weight
        
        # 基于类型加权
        type_weights = {
            'github_release': 30,
            'pricing': 40,
            'funding': 50,
            'partnership': 45,
            'product_update': 35
        }
        score += type_weights.get(item.get('type', ''), 10)
        
        return min(score, 100)
    
    def deduplicate(self, items):
        """内容去重"""
        if len(items) < 2:
            return items
        
        # 简单相似度去重
        contents = [item.get('content', '') for item in items]
        vectorizer = TfidfVectorizer()
        tfidf_matrix = vectorizer.fit_transform(contents)
        
        unique_items = []
        seen_indices = set()
        
        for i in range(len(items)):
            if i in seen_indices:
                continue
                
            unique_items.append(items[i])
            
            # 标记相似项
            for j in range(i + 1, len(items)):
                similarity = cosine_similarity(tfidf_matrix[i:i+1], tfidf_matrix[j:j+1])[0][0]
                if similarity > 0.8:  # 相似度阈值
                    seen_indices.add(j)
        
        return unique_items
    
    def process(self, raw_updates):
        """处理所有更新"""
        processed = []
        
        for item in raw_updates:
            # 计算重要性
            item['importance_score'] = self.calculate_importance(item)
            
            # 提取关键词
            if 'content' in item:
                item['keywords'] = self.extract_keywords(item['content'])
            
            # 分类
            item['category'] = self._classify(item)
            
            processed.append(item)
        
        # 去重
        processed = self.deduplicate(processed)
        
        # 按重要性排序
        processed.sort(key=lambda x: x['importance_score'], reverse=True)
        
        return processed
```

### 2.6 通知模块 (wecom_notifier.py)
```python
import requests

class WecomNotifier:
    def __init__(self, webhook_url):
        self.webhook_url = webhook_url
    
    def send_summary(self, updates, report_path):
        """发送摘要通知"""
        high_importance = [u for u in updates if u['importance_score'] >= 60]
        medium_importance = [u for u in updates if 30 <= u['importance_score'] < 60]
        
        message = {
            "msgtype": "markdown",
            "markdown": {
                "content": f"""
## 📊 竞品监控周报摘要（{len(updates)}条更新）

### 🔥 重要更新（{len(high_importance)}条）
{self._format_items(high_importance[:3])}

### 📋 一般更新（{len(medium_importance)}条）
{self._format_items(medium_importance[:5])}

📄 [完整报告下载]({report_path})
                """
            }
        }
        
        response = requests.post(self.webhook_url, json=message)
        return response.status_code == 200
    
    def _format_items(self, items):
        if not items:
            return "> 暂无"
        
        lines = []
        for item in items:
            lines.append(f"> • **{item['competitor']}**: {item.get('notes', item.get('latest_commit', '页面更新'))[:50]}...")
        
        return '\n'.join(lines)
```

| 指标 | 告警阈值 | 通知方式 |

## 五、运维要求

- **日志保留**：3个月
- **数据备份**：每日备份数据库
- **故障恢复**：脚本失败自动重试3次，失败告警
- **性能监控**：单次执行时间≤30分钟
- **人工审核**：每周一上午人工复核自动报告
